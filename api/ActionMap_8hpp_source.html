48  // Number of repeats that happened after holding for this long: -1 before the first one. The small
49  // tolerance keeps a repeat that falls exactly on a frame: in float, (0.5 - 0.4) / 0.1 gives 0.9999999
50  [[nodiscard]] float repeatCount(float heldDuration) const
51  {
52  constexpr float tolerance = 1e-4f;
53  if (heldDuration + tolerance < _repeatDelay) {
54  return -1.0f;
55  }
56  return std::floor((heldDuration - _repeatDelay) / _repeatInterval + tolerance);
57  }

114  void update(const InputSource &source, float elapsed = 0.0f)
115  {
116  for (auto &[action, state] : _actions) {
117  state.wasActive = state.active;
118  state.strength = 0.0f;
119  for (const Binding &binding : state.bindings) {
120  state.strength = std::max(state.strength, binding.strength(source, _deadZone));
121  }
122  state.active = state.strength > 0.0f;
123 
124  if (!state.active) {
125  // On the update where it is released, the action keeps how long it was held
126  if (!state.wasActive) {
127  state.heldDuration = 0.0f;
128  }
129  state.repeated = false;
130  } else if (!state.wasActive) {
131  // Just pressed: the press itself counts as the first repeat
132  state.heldDuration = 0.0f;
133  state.repeated = true;
134  } else {
135  const float before = state.heldDuration;
136  state.heldDuration += elapsed;
137  state.repeated = repeatCount(state.heldDuration) > repeatCount(before);
138  }
139  }
