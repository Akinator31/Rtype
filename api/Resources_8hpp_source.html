109 // Owns a 3D model: its meshes, materials and textures, and the animations of its skeleton if it has one. A
110 // moved-from model frees nothing.
111 class Model
112 {
113 private:
114  // raylib samples the animations of glTF files at this rate
115  static constexpr float framesPerSecond = 60.0f;
116 
117  ::Model _model;
118  ModelAnimation *_animations = nullptr;
119  int _animationCount = 0;
120 
121  // IsModelValid() rejects every skinned model in raylib 6.0: it requires the bone buffers in graphics
122  // memory, which raylib only uploads with GPU skinning, disabled by default (it skins on the CPU instead).
123  // A model is usable once its meshes and materials are loaded and its vertices are uploaded.
124  static bool isLoaded(const ::Model &model)
125  {
126  if (model.meshes == nullptr || model.materials == nullptr || model.meshCount <= 0 ||
127  model.materialCount <= 0) {
128  return false;
129  }
130  for (int index = 0; index < model.meshCount; ++index) {
131  if (model.meshes[index].vboId == nullptr || model.meshes[index].vboId[0] == 0) {
132  return false;
133  }
134  }
135  return true;
136  }
137 
138  // Models are mostly seen smaller than their textures: without mipmaps, colours and relief shimmer.
139  // Materials without a texture share raylib's default one, which is left alone.
140  static void smoothTexture(Texture2D &texture)
141  {
142  if (IsTextureValid(texture) && texture.id != rlGetTextureIdDefault()) {
143  GenTextureMipmaps(&texture);
144  SetTextureFilter(texture, TEXTURE_FILTER_TRILINEAR);
145  }
146  }
147 
148 public:
149  explicit Model(const std::filesystem::path &file) : _model(LoadModel(file.string().c_str()))
150  {
151  if (!isLoaded(_model)) {
152  UnloadModel(_model);
153  throw std::runtime_error("Cannot load model " + file.string());
154  }
155  for (int index = 0; index < _model.materialCount; ++index) {
156  smoothTexture(_model.materials[index].maps[MATERIAL_MAP_ALBEDO].texture);
157  smoothTexture(_model.materials[index].maps[MATERIAL_MAP_NORMAL].texture);
158  }
159  if (_model.skeleton.boneCount > 0) {
160  _animations = LoadModelAnimations(file.string().c_str(), &_animationCount);
161  }
162  }
163  ~Model()
164  {
165  if (_animations != nullptr) {
166  UnloadModelAnimations(_animations, _animationCount);
167  }
168  if (isLoaded(_model)) {
169  UnloadModel(_model);
170  }
171  }

189  // Index of the animation with this name, or -1 if the model has none by this name.
190  [[nodiscard]] int findAnimation(const std::string &name) const
191  {
192  for (int index = 0; index < _animationCount; ++index) {

194  return index;
195  }

200  // Moves the vertices of the model to their place at a time of an animation, in seconds. The animation
201  // loops. The model keeps this pose until it is animated again. Const like drawing: only the vertices in
202  // graphics memory change, and the renderer, which poses models, only reads the resources.
203  void animate(int animation, float time) const
204  {
205  if (hasAnimation(animation)) {
206  UpdateModelAnimation(_model, _animations[animation], time * framesPerSecond);
207  }
208  }
209 
210  // Like animate(), with a pose between two animations: `amount` goes from 0 for the first one to 1 for the
211  // second one.
212  void blendAnimations(int first, float firstTime, int second, float secondTime, float amount) const
213  {
214  if (!hasAnimation(first) || !hasAnimation(second)) {
215  animate(second, secondTime);
216  return;
217  }
218  UpdateModelAnimationEx(_model, _animations[first], firstTime * framesPerSecond, _animations[second],
219  secondTime * framesPerSecond, std::clamp(amount, 0.0f, 1.0f));
220  }
